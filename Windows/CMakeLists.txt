cmake_minimum_required(VERSION 3.20)
project(ARoseScreenSaver VERSION 1.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(arose-engine-tests ../Tests/Parity/main.cpp)
add_executable(arose-rendering-tests ../Tests/WindowsRendering/main.cpp)
add_executable(arose-figure-tests ../Tests/FigureState/main.cpp)
enable_testing()
add_test(NAME engine COMMAND arose-engine-tests --self-test)
add_test(NAME rendering-data COMMAND arose-rendering-tests)
add_test(NAME saved-figure COMMAND arose-figure-tests)
if(WIN32)
    enable_language(RC)
    add_executable(arose-image-tests ../Tests/WindowsRendering/images.cpp)
    target_link_libraries(arose-image-tests PRIVATE d2d1 windowscodecs ole32)
    add_test(NAME rendering-images COMMAND arose-image-tests)
    add_executable(arose-gpu-tests ../Tests/WindowsRendering/gpu.cpp)
    target_link_libraries(arose-gpu-tests PRIVATE d3d11 dxgi d3dcompiler d2d1 windowscodecs ole32)
    add_test(NAME rendering-gpu COMMAND arose-gpu-tests)
    set_tests_properties(rendering-gpu PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120)
    add_executable(ARose WIN32 main.cpp ARose.rc)
    set_target_properties(ARose PROPERTIES SUFFIX ".scr")
    target_include_directories(ARose PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(ARose PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_link_libraries(ARose PRIVATE d2d1 dwmapi dwrite comctl32 shell32 advapi32 user32 gdi32 d3d11 dxgi d3dcompiler windowscodecs ole32)
    if(MSVC)
        # ARose.rc embeds the complete application manifest as resource 1.
        # Suppress MSVC's generated manifest to avoid duplicate MANIFEST input.
        target_link_options(ARose PRIVATE /MANIFEST:NO)
        target_compile_options(ARose PRIVATE /W4 /utf-8 /fp:precise)
        target_compile_options(arose-engine-tests PRIVATE /W4 /utf-8 /fp:precise)
        target_compile_options(arose-rendering-tests PRIVATE /W4 /utf-8 /fp:precise)
        target_compile_options(arose-figure-tests PRIVATE /W4 /utf-8 /fp:precise)
        target_compile_options(arose-image-tests PRIVATE /W4 /utf-8 /fp:precise)
        target_compile_options(arose-gpu-tests PRIVATE /W4 /utf-8 /fp:precise)
        set_property(TARGET ARose arose-engine-tests arose-rendering-tests arose-figure-tests arose-image-tests arose-gpu-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(ARose PRIVATE -Wall -Wextra -ffp-contract=off)
        target_link_options(ARose PRIVATE -municode -static)
    endif()
    add_test(NAME window-smoke COMMAND ARose /t)
    set_tests_properties(window-smoke PROPERTIES TIMEOUT 15)
endif()
